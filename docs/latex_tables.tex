% Robot Interface Pseudocode
% --------------------------------------------
\begin{algorithm}[h]
\caption{Robot Interface: Goal and Timing Prediction}
\begin{algorithmic}[1]
\REQUIRE Topology graph $\mathcal{G}$, Current exhibit $e_t$, Gaze history $\mathcal{H}_t$
\ENSURE Goal distribution $p(g|context)$, Dwell prediction $\hat{d}$, Readiness $r$

\STATE context $\gets$ ContextBuilder.build($e_t$, $\mathcal{H}_t$, $\mathcal{G}$)
\STATE mem $\gets$ MemoryManager.get\_context($e_t$, $\mathcal{H}_t$)
\STATE prediction $\gets$ LLMReasoner.predict(context, mem)
\STATE trajectory $\gets$ SequencePredictor.rollout(context, mem, $K=5$)

\RETURN \{
    \texttt{goal\_dist}: $p(g|context)$,
    \texttt{dwell}: prediction.dwell,
    \texttt{readiness}: prediction.attention,
    \texttt{trajectory}: trajectory
\}
\end{algorithmic}
\end{algorithm}
